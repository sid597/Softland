## Run C: the markdown files: headings per file, the size of each piece when cut at every heading, and repeated headings (a line heuristic)

;; HEURISTIC: a heading is a line of 1-6 # outside ``` or ~~~ fences; a leaf section runs from a heading to the next heading of any level
(def md (filter #(re-find #"\.md$" %) paths))
(defn headings [text]
  (let [lines (vec (str/split text #"\n" -1))]
    (loop [i 0 fence false out []]
      (if (>= i (count lines)) {:lines lines :hs out}
        (let [l (lines i)]
          (cond (re-find #"^\s{0,3}(```|~~~)" l) (recur (inc i) (not fence) out)
                fence (recur (inc i) fence out)
                :else (if-let [[_ hs t] (re-find #"^\s{0,3}(#{1,6})\s+(.*?)\s*#*\s*$" l)]
                        (recur (inc i) fence (conj out {:i i :level (count hs) :title t}))
                        (recur (inc i) fence out))))))))
(def stats (for [x md :let [{:keys [lines hs]} (headings (texts x))
                            starts (map :i hs) ends (concat (rest starts) [(count lines)])
                            leaf (map (fn [s e] (utf8 (str/join "\n" (subvec lines s e)))) starts ends)
                            pre (if (seq hs) (utf8 (str/join "\n" (subvec lines 0 (first starts)))) (utf8 (texts x)))
                            hpaths (loop [hs hs stack [] out []] (if-let [h (first hs)] (let [st (conj (vec (remove #(>= (:level %) (:level h)) stack)) h)] (recur (rest hs) st (conj out (mapv :title st)))) out))]]
             {:path x :n (count hs) :maxleaf (if (seq leaf) (apply max leaf) 0) :pre pre :size (utf8 (texts x))
              :dup-title (- (count hs) (count (set (map :title hs)))) :dup-path (- (count hpaths) (count (set hpaths)))}))
(println "md files" (count stats) "| with no heading:" (count (filter #(zero? (:n %)) stats)) "| headings:" (reduce + (map :n stats)) "| most in one file:" (apply max (map :n stats)))
(println "files with more than 63 headings:" (count (filter #(> (:n %) 63) stats)) "| more than 127:" (count (filter #(> (:n %) 127) stats)) "| more than 255:" (count (filter #(> (:n %) 255) stats)))
(println "md files over 65536 bytes:" (count (filter #(> (:size %) 65536) stats)) "| of them, still a piece over the bound when cut at every heading:" (count (filter #(and (> (:size %) 65536) (or (> (:maxleaf %) 65536) (> (:pre %) 65536))) stats)) "| (a leaf section over:" (count (filter #(> (:maxleaf %) 65536) stats)) "files; text before the first heading over:" (count (filter #(> (:pre %) 65536) stats)) "files)")
(println "files where a heading title repeats:" (count (filter #(pos? (:dup-title %)) stats)) "| where a whole heading path repeats:" (count (filter #(pos? (:dup-path %)) stats)))

;; HEURISTIC tiling: cut before every line that begins with ( except the first; each part keeps the lines up to the next cut
(defn tile-clj [text]
  (let [lines (vec (str/split text #"(?<=\n)"))
        starts (sort (keep-indexed (fn [i l] (when (re-find #"^\(" l) i)) lines))
        cuts (vec (concat [0] (rest starts) [(count lines)]))]
    (map (fn [a b] (apply str (subvec lines a b))) cuts (rest cuts))))
